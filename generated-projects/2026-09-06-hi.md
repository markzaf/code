import React from "react";
type Props = React.ButtonHTMLAttributes<HTMLButtonElement> & {
  variant?: "primary" | "ghost" | "danger";
  size?: "sm" | "md";
};

export default function Button({ variant = "primary", size = "md", className = "", children, ...props }: Props) {
  const base = "inline-flex items-center justify-center font-medium rounded-md transition";
  const sizes = {
    sm: "px-3 py-1.5 text-sm",
    md: "px-4 py-2 text-sm"
  };
  const variants: Record<string, string> = {
    primary: "bg-brand-500 text-white hover:bg-brand-600",
    ghost: "bg-white border border-slate-200 hover:bg-slate-50",
    danger: "bg-red-600 text-white hover:bg-red-700"
  };
  return (
    <button className={`${base} ${sizes[size]} ${variants[variant]} ${className}`} {...props}>
      {children}
    </button>
  );
}

// components/Board/BoardView.tsx
"use client";
import React, { useState } from "react";
import { DragDropContext, Droppable } from "@hello-pangea/dnd";
import Column from "./Column";
import { BoardState, Column as ColumnType } from "@/lib/data";

export default function BoardView({ initial }: { initial: BoardState }) {
  const [board, setBoard] = useState<BoardState>(initial);
  const [activeTask, setActiveTask] = useState<any | null>(null);

  const onDragEnd = (result: any) => {
    const { source, destination, type } = result;
    if (!destination) return;

    if (type === "COLUMN") {
      const newCols = Array.from(board.columns);
      const [moved] = newCols.splice(source.index, 1);
      newCols.splice(destination.index, 0, moved);
      setBoard({ columns: newCols });
      return;
    }

    // task move
    const startColIdx = board.columns.findIndex((c) => c.id === source.droppableId);
    const endColIdx = board.columns.findIndex((c) => c.id === destination.droppableId);
    const startCol = board.columns[startColIdx];
    const endCol = board.columns[endColIdx];

    if (startCol === endCol) {
      const newTasks = Array.from(startCol.tasks);
      const [moved] = newTasks.splice(source.index, 1);
      newTasks.splice(destination.index, 0, moved);
      const newCol: ColumnType = { ...startCol, tasks: newTasks };
      const newCols = Array.from(board.columns);
      newCols[startColIdx] = newCol;
      setBoard({ columns: newCols });
    } else {
      const startTasks = Array.from(startCol.tasks);
      const [moved] = startTasks.splice(source.index, 1);
      const endTasks = Array.from(endCol.tasks);
      endTasks.splice(destination.index, 0, moved);
      const newCols = Array.from(board.columns);
      newCols[startColIdx] = { ...startCol, tasks: startTasks };
      newCols[endColIdx] = { ...endCol, tasks: endTasks };
      setBoard({ columns: newCols });
    }
  };

  return (
    <div className="px-4 py-6">
      <DragDropContext onDragEnd={onDragEnd}>
        <Droppable droppableId="board" direction="horizontal" type="COLUMN">
          {(provided) => (
            <div ref={provided.innerRef} {...provided.droppableProps} className="flex gap-4 overflow-x-auto pb-6">
              {board.columns.map((col, idx) => (
                <Column key={col.id} column={col} index={idx} onOpen={setActiveTask} />
              ))}
              {provided.placeholder}
              <div className="w-80 flex-shrink-0">
                <button className="w-full h-10 rounded-md border border-dashed border-slate-200 bg-white text-sm text-slate-600">+ Add column</button>
              </div>
            </div>
          )}
        </Droppable>
      </DragDropContext>

      {activeTask && (
        <div role="dialog" aria-modal="true" className="fixed inset-0 z-40 flex items-center justify-center">
          <div className="absolute inset-0 bg-black/40" onClick={() => setActiveTask(null)} />
          <div className="relative z-50 w-full max-w-2xl bg-white rounded-lg p-6 shadow-lg">
            <div className="flex items-start justify-between gap-4">
              <div>
                <h2 className="text-lg font-semibold">{activeTask.title}</h2>
                <p className="text-sm text-slate-600 mt-2">{activeTask.description}</p>
              </div>
              <div>
                <button onClick={() => setActiveTask(null)} className="text-slate-500 hover:text-slate-700">Close</button>
              </div>
            </div>
            <div className="mt-4 text-sm text-slate-500">
              <div>Due: {activeTask.due ? new Date(activeTask.due).toLocaleDateString() : "No due date"}</div>
              <div className="mt-2">Labels: {activeTask.labels?.join(", ") || "—"}</div>
            </div>
          </div>
        </div>
      )}
    </div>
  );
}

Conceptual SQL (Postgres) equivalent (short)
- users(id PK, uuid, email UNIQUE, name, password, avatar, created_at, updated_at)
- workspaces(id PK, uuid, name, owner_id FK -> users.id, created_at, updated_at)
- workspace_members(id PK, workspace_id FK, user_id FK, role, created_at)
- boards(id PK, uuid, title, workspace_id FK, created_at, updated_at)
- columns(id PK, uuid, title, board_id FK, position int)
- tasks(id PK, uuid, title, description, labels text[], due timestamp, assignee_id FK -> users.id, column_id FK, position int, created_at, updated_at)

Notes:
- labels are stored as a String[] Postgres array in Prisma.
- position fields are integer ordering keys; moves are implemented via transactions.

2) API SPEC

Base URL: /api

Auth
- POST /api/auth/register
  - Request: { name?: string, email: string, password: string }
  - Response: { user: { id, uuid, email, name }, token: string }
- POST /api/auth/login
  - Request: { email: string, password: string }
  - Response: { user: { id, uuid, email, name }, token: string }
- GET /api/auth/me
  - Header: Authorization: Bearer <token>
  - Response: { user }

Workspace & Board
- GET /api/workspaces
  - Auth required
  - Returns list of workspaces the user is a member of
- POST /api/workspaces
  - Body: { name: string }
  - Creates workspace (owner = current user)
- GET /api/boards/:boardId
  - Auth required; returns board with columns and tasks (columns sorted by position, tasks sorted by position)
  - Response: { board: { id, uuid, title, columns: [ { id, uuid, title, position, tasks: [...] } ] } }
- POST /api/boards
  - Body: { workspaceId: number, title: string }
  - Create board and default columns

Columns
- POST /api/boards/:boardId/columns
  - Auth; Body: { title: string, position?: number }
  - Response: created column
- PATCH /api/columns/:columnId
  - Body: { title?: string, position?: number }
  - Update column
- DELETE /api/columns/:columnId
  - Delete column and optionally move/delete tasks

Tasks
- POST /api/columns/:columnId/tasks
  - Body: { title: string, description?: string, labels?: string[], due?: string (ISO), assigneeId?: number, position?: number }
  - Creates a task in column
  - Response: { task }
- PATCH /api/tasks/:taskId
  - Body: partial task fields to update
  - Response: updated task
- DELETE /api/tasks/:taskId
  - Delete task
- POST /api/tasks/move
  - Body: { taskId: number, toColumnId: number, toPosition: number }
  - Moves task between columns or reorder within column; server updates positions of other tasks in a transaction
  - Response: { ok: true }

All endpoints that modify data require Authorization header and validate membership on workspace/board.

// backend/src/middleware/auth.ts
import { Request, Response, NextFunction } from "express";
import jwt from "jsonwebtoken";
import prisma from "../prisma";
import dotenv from "dotenv";
dotenv.config();

const JWT_SECRET = process.env.JWT_SECRET || "dev-secret";

export async function requireAuth(req: Request, res: Response, next: NextFunction) {
  try {
    const auth = req.headers.authorization;
    if (!auth || !auth.startsWith("Bearer ")) {
      return res.status(401).json({ error: "Missing Authorization" });
    }
    const token = auth.split(" ")[1];
    const payload = jwt.verify(token, JWT_SECRET) as any;
    if (!payload?.userId) return res.status(401).json({ error: "Invalid token" });
    // optional: load user from DB
    const user = await prisma.user.findUnique({ where: { id: payload.userId } });
    if (!user) return res.status(401).json({ error: "User not found" });
    req.user = { id: user.id, email: user.email, name: user.name ?? undefined };
    next();
  } catch (err) {
    console.error("Auth error", err);
    return res.status(401).json({ error: "Unauthorized" });
  }
}

// backend/src/routes/auth.ts
import { Router } from "express";
import prisma from "../prisma";
import bcrypt from "bcrypt";
import jwt from "jsonwebtoken";
import dotenv from "dotenv";
dotenv.config();

const router = Router();
const JWT_SECRET = process.env.JWT_SECRET || "dev-secret";
const SALT_ROUNDS = parseInt(process.env.BCRYPT_SALT_ROUNDS || "10", 10);

router.post("/register", async (req, res) => {
  const { name, email, password } = req.body;
  if (!email || !password) return res.status(400).json({ error: "email and password are required" });

  const existing = await prisma.user.findUnique({ where: { email } });
  if (existing) return res.status(409).json({ error: "Email already in use" });

  const hashed = await bcrypt.hash(password, SALT_ROUNDS);
  const user = await prisma.user.create({ data: { name, email, password: hashed } });

  // create default workspace for user
  const workspace = await prisma.workspace.create({
    data: {
      name: `${name ?? "User"}'s Workspace`,
      ownerId: user.id,
      members: { create: { userId: user.id, role: "owner" } },
      boards: {
        create: {
          title: "Project Board",
          columns: {
            createMany: {
              data: [
                { title: "To Do", position: 0 },
                { title: "In Progress", position: 1 },
                { title: "Done", position: 2 }
              ]
            }
          }
        }
      }
    },
    include: { boards: { include: { columns: true } } }
  });

  const token = jwt.sign({ userId: user.id }, JWT_SECRET, { expiresIn: "7d" });

  res.status(201).json({ user: { id: user.id, uuid: user.uuid, email: user.email, name: user.name }, token, workspaceId: workspace.id });
});

router.post("/login", async (req, res) => {
  const { email, password } = req.body;
  if (!email || !password) return res.status(400).json({ error: "email and password are required" });

  const user = await prisma.user.findUnique({ where: { email } });
  if (!user) return res.status(401).json({ error: "Invalid credentials" });

  const ok = await bcrypt.compare(password, user.password);
  if (!ok) return res.status(401).json({ error: "Invalid credentials" });

  const token = jwt.sign({ userId: user.id }, JWT_SECRET, { expiresIn: "7d" });

  res.json({ user: { id: user.id, uuid: user.uuid, email: user.email, name: user.name }, token });
});

    return res.status(401).json({ error: "Missing Authorization" });

    res.json({ user: { id: user.id, uuid: user.uuid, email: user.email, name: user.name } });

// create column in board
router.post("/board/:boardId/columns", requireAuth, async (req, res) => {
  const boardId = Number(req.params.boardId);
  const { title, position } = req.body;
  if (!title) return res.status(400).json({ error: "title required" });

  // membership check
  const board = await prisma.board.findUnique({ where: { id: boardId } });
  if (!board) return res.status(404).json({ error: "Board not found" });

  if (!mem) return res.status(403).json({ error: "Forbidden" });

  // Determine position
  const maxPosRes = await prisma.column.aggregate({ where: { boardId }, _max: { position: true } });
  const nextPos = (maxPosRes._max.position ?? -1) + 1;

import { Router } from "express";
import prisma from "../prisma";
import { requireAuth } from "../middleware/auth";

const router = Router();

// create task in a column
router.post("/columns/:columnId/tasks", requireAuth, async (req, res) => {
  const columnId = Number(req.params.columnId);
  const { title, description, labels, due, assigneeId, position } = req.body;
  if (!title) return res.status(400).json({ error: "title required" });

  const column = await prisma.column.findUnique({ where: { id: columnId }, include: { board: true } });
  if (!column) return res.status(404).json({ error: "Column not found" });

  const mem = await prisma.workspaceMember.findFirst({ where: { workspaceId: column.board.workspaceId, userId: req.user!.id } });
  if (!mem) return res.status(403).json({ error: "Forbidden" });

  const maxPosRes = await prisma.task.aggregate({ where: { columnId }, _max: { position: true } });
  const nextPos = (maxPosRes._max.position ?? -1) + 1;

  const task = await prisma.task.create({
    data: {
      title,
      description,
      labels: labels ?? [],
      due: due ? new Date(due) : undefined,
      assigneeId: assigneeId ? Number(assigneeId) : undefined,
      columnId,
      position: typeof position === "number" ? position : nextPos
    },
    include: { assignee: true }
  });

  res.status(201).json({ task });
});

// update task
router.patch("/:taskId", requireAuth, async (req, res) => {
  const taskId = Number(req.params.taskId);
  const body = req.body;
  const task = await prisma.task.findUnique({ where: { id: taskId }, include: { column: { include: { board: true } } } });
  if (!task) return res.status(404).json({ error: "Task not found" });

  const mem = await prisma.workspaceMember.findFirst({ where: { workspaceId: task.column.board.workspaceId, userId: req.user!.id } });
  if (!mem) return res.status(403).json({ error: "Forbidden" });

  const data: any = {};
  if (body.title !== undefined) data.title = body.title;
  if (body.description !== undefined) data.description = body.description;
  if (body.labels !== undefined) data.labels = body.labels;
  if (body.due !== undefined) data.due = body.due ? new Date(body.due) : null;
  if (body.assigneeId !== undefined) data.assigneeId = body.assigneeId;
  const updated = await prisma.task.update({ where: { id: taskId }, data, include: { assignee: true } });
  res.json({ task: updated });
});

// delete task
router.delete("/:taskId", requireAuth, async (req, res) => {
  const taskId = Number(req.params.taskId);
  const task = await prisma.task.findUnique({ where: { id: taskId }, include: { column: { include: { board: true } } } });
  if (!task) return res.status(404).json({ error: "Task not found" });
  const mem = await prisma.workspaceMember.findFirst({ where: { workspaceId: task.column.board.workspaceId, userId: req.user!.id } });
  if (!mem) return res.status(403).json({ error: "Forbidden" });
  await prisma.task.delete({ where: { id: taskId } });
  res.json({ ok: true });
});

// move task (reorder between columns or within same column)
router.post("/move", requireAuth, async (req, res) => {
  /*
    body: { taskId: number, toColumnId: number, toPosition: number }
    Server will:
      - verify permissions
      - remove gap from source column (decrement positions > source position)
      - make space in target column (increment positions >= toPosition)
      - update task.columnId and position
  */
  const { taskId, toColumnId, toPosition } = req.body;
  if (!taskId || toColumnId === undefined || toPosition === undefined) {
    return res.status(400).json({ error: "taskId, toColumnId and toPosition required" });
  }
  const tid = Number(taskId);
  const toColId = Number(toColumnId);
  const desiredPos = Number(toPosition);

  const task = await prisma.task.findUnique({ where: { id: tid }, include: { column: { include: { board: true } } } });
  if (!task) return res.status(404).json({ error: "Task not found" });
  const fromCol = task.column;
  const board = fromCol.board;

  // permission check
  const mem = await prisma.workspaceMember.findFirst({ where: { workspaceId: board.workspaceId, userId: req.user!.id } });
  if (!mem) return res.status(403).json({ error: "Forbidden" });

  // ensure target column is on same board
  const toColumn = await prisma.column.findUnique({ where: { id: toColId }, include: { board: true } });
  if (!toColumn) return res.status(404).json({ error: "Destination column not found" });
  if (toColumn.boardId !== board.id) return res.status(400).json({ error: "Columns belong to different boards" });

  await prisma.$transaction(async (tx) => {
    // close gap in source column
    await tx.task.updateMany({
      where: { columnId: fromCol.id, position: { gt: task.position } },
      data: { position: { decrement: 1 } }
    });

    // make space in destination
    await tx.task.updateMany({
      where: { columnId: toColId, position: { gte: desiredPos } },
      data: { position: { increment: 1 } }
    });

    // move task
    await tx.task.update({
      where: { id: tid },
      data: { columnId: toColId, position: desiredPos }
    });
  });

  res.json({ ok: true });
});

export default router;

D. Auth & token storage: recommended approaches
Two strategies:

1) Short-term/easy: client stores JWT in memory or localStorage and sends Authorization header:
   - Pros: easy to implement w/ fetch.
   - Cons: vulnerable to XSS if token stored in localStorage.

2) Recommended/secure: use HttpOnly, Secure cookie for access token or use refresh-token cookie pattern:
   - Store tokens in HttpOnly cookies set by the backend on login (Set-Cookie) to avoid JS access.
   - For Next.js frontend, you can either:
     - Proxy API through Next server routes (Next API routes) and add cookie forwarding, or
     - Add CORS and credentials: app should call backend with fetch(..., { credentials: 'include' }) and backend must set Access-Control-Allow-Credentials header and set Secure, HttpOnly and SameSite attributes on cookies.
   - Pros: protects against XSS token theft.
   - Cons: slightly more server-side wiring (cookie set endpoint, CSRF consideration).

If you keep the simple Authorization approach, ensure:
- Use strong JWT_SECRET,
- Use short expiry and refresh token pattern if needed,
- Guard sensitive actions by server-side checks.

Example fetch wrapper (client-side, localStorage token pattern):
```ts
// lib/api.ts
const API = process.env.NEXT_PUBLIC_API_URL;

export async function apiFetch(path: string, opts: RequestInit = {}, token?: string) {
  const headers: Record<string, string> = {
    "Content-Type": "application/json",
    ...(opts.headers as Record<string,string> || {})
  };
  if (token) headers["Authorization"] = `Bearer ${token}`;
  const res = await fetch(`${API}${path}`, { ...opts, headers });
  if (!res.ok) {
    const err = await res.json().catch(() => ({ message: res.statusText }));
    throw err;
  }
  return res.json();
}

C. Step-by-step go-live (example: Frontend on Vercel, Backend on Railway, Postgres on Railway / Supabase)
1) Prepare production resources:
   - Create a production Postgres instance (Supabase / Neon / Railway) and note DATABASE_URL.
   - Create backend host (Railway, Render or Fly). If using Railway, connect the Postgres service to the backend project.

2) Backend deployment:
   - Push backend code to repo and create Railway project (or Render service).
   - Add environment variables:
     - DATABASE_URL → from Postgres provider
     - JWT_SECRET → strong random string
     - BCRYPT_SALT_ROUNDS → 10
     - PORT → 4000 (or provider default)
     - CORS_ORIGIN → https://your-frontend-domain.com
   - On first deploy, run migrations:
     - Either via Railway migration UI or run `npx prisma migrate deploy` from your container/CI job.
     - Verify tables created; optionally run seed script to create sample user/workspace/board.

   - Verify API health:
     - GET https://api.yourdomain.com/health → { status: "ok" }

3) Frontend deployment:
   - Connect GitHub repo to Vercel, configure production branch (main).
   - Set environment variable NEXT_PUBLIC_API_URL to https://api.yourdomain.com (or the backend domain).
   - Deploy and verify site loads, links to /dashboard route.
   - If using HttpOnly cookie authentication, ensure cross-site cookie config and CORS credentials alignment between frontend and backend (cors credentials true, fetch credentials:'include').

4) DNS & TLS:
   - Point your domain to Vercel (frontend) via provider docs; Vercel issues TLS automatically.
   - Point API domain to backend provider; enable TLS (Railway/Render do automatically for custom domains).

5) Smoke tests:
   - Register a new user (signup), login, create a task, move a task between columns — verify DB entries via backend dashboards or psql.
   - Run automated smoke test script in CI.

6) Monitoring & backups:
   - Enable error reporting (Sentry) on backend and frontend.
   - Enable DB backups & point to storage for retention.
   - Setup alerts (uptime, error rate, latency) using provider or external monitoring.

7) Rollback plan:
   - Keep last working build in provider (Vercel/Render keep versions).
   - DB migrations: use non-destructive migrations first; or snapshot DB before running a destructive migration so you can restore.

D. Running Prisma migrations in CI/production
- Use `npx prisma migrate deploy` (not migrate dev) in CI/deploy pipeline to apply migrations securely.
- Add a step in your deploy process to run `npx prisma generate` then `npx prisma migrate deploy`.
- For initial setup locally: `npx prisma migrate dev --name init` creates migration files.

E. Secrets & key rotation
- Rotate JWT_SECRET periodically and expire tokens after rotation.
- Store secrets in the hosting provider’s secret manager. Avoid secrets in repo.

F. Optional: enabling realtime updates
- Add socket.io or server-sent events:
  - Host a socket.io server attached to your Express app or deploy a separate service (e.g., on Fly/Railway).
  - On task move/create/update, broadcast board updates to members of the workspace so clients can reconcile state.
- Security: auth sockets via token handshake and validate membership.

G. Backups, logging, and SLOs
- Backups: enable daily DB snapshots, weekly full retention, point-in-time recovery if available.
- Logs: forward backend logs to provider log drain (Papertrail/Datadog).
- SLO suggestions:
  - Availability: 99.9% (depending on SLA)
  - API latency p95 < 500ms

// inside BoardView.onDragEnd
const token = localStorage.getItem("token"); // or use cookie flow

  });
```

B. Login flow with cookie pattern (backend & frontend)
- Backend (login) returns Set-Cookie header:
  - Use express cookie setting:
    - res.cookie("token", token, { httpOnly: true, secure: true, sameSite: 'lax', maxAge: 7*24*3600*1000 })
- Frontend calls login endpoint with credentials: fetch('/api/auth/login', { method:'POST', credentials:'include', body:... })
- For subsequent requests, call fetch with credentials: 'include' and backend must allow credentials.
